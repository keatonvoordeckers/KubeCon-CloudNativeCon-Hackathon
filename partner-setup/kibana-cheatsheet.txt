# ---------------------------------------------------------------------
# 3. SEARCH + FILTERS: the shape of every /search call
#    "must"   = affects ranking (the search text)
#    "filter" = yes/no only, faster, cached (dropdowns + checkboxes)
#    Topic filter checks the official track OR the inferred one.
# ---------------------------------------------------------------------
GET kubecon-talks/_search
{
  "_source": ["title", "start", "location.venue", "location.room", "registration.type"],
  "query": {
    "bool": {
      "must": [
        { "multi_match": { "query": "agents", "fields": ["title^3", "abstract"], "fuzziness": "AUTO" } }
      ],
      "filter": [
        { "term": { "day": "2026-11-10" } },
        { "bool": { "should": [
            { "term": { "topics": "AI Inference + Agentic" } },
            { "term": { "topics_inferred": "AI Inference + Agentic" } }
          ], "minimum_should_match": 1 } },
        { "terms": { "registration.type": ["kubecon_pass", "free_open"] } },
        { "term": { "accessibility.wheelchair_accessible": true } }
      ]
    }
  }
}

# TIPS
# - Trim noisy output: add ?filter_path=hits.hits._source,aggregations to the URL
#     GET kubecon-talks/_search?filter_path=aggregations
# - "term" needs the EXACT value ("Salt Palace", not "salt palace"). Use section 1 to see exact values.
# - Use "term" on keyword fields (venue, level, room, topics, day...).
#   Use "match"/"multi_match" on text fields (title, abstract, speakers.name).
# - Why is my agg empty? Check the query part first. Aggs only count matching docs.
# - In Python (inside FastAPI), the same JSON goes in as keyword arguments:
#     es.search(index="kubecon-talks", query={...}, aggs={...}, size=20, sort=[...])
#   then read  resp["hits"]["hits"]  and  resp["aggregations"]["venues"]["buckets"]
